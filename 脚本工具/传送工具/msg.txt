<select id="queryList" resultMap="BaseResultMap">
    SELECT
        id,
        company_id,
        report_period,
        report_date,
        related_party_name,
        related_party_relation,
        project_one,
        project_two,
        project_three,
        amount,
        proportion,
        delete_identify
    FROM faf_related_party frp
    WHERE frp.delete_identify = 0
      AND company_id = #{companyId}
      AND report_period &gt;= CAST(CONCAT(#{startYear}, '11') AS INT)
      AND report_period &lt;= CAST(CONCAT(#{endYear}, '41') AS INT)

    <choose>
        <!-- 1. 选择“年报”：只要第四季度 (Q4)，即结尾是 41 -->
        <when test="reportType == '年报'">
            AND CAST(report_period AS VARCHAR) LIKE '%41'
        </when>

        <!-- 2. 选择“半年报”：只要第二季度 (Q2)，即结尾是 21 -->
        <when test="reportType == '半年报'">
            AND CAST(report_period AS VARCHAR) LIKE '%21'
        </when>

        <!-- 3. 选择“最新”：今年取最新一季，历年取年报(Q4) -->
        <when test="reportType == '最新'">
            AND frp.id IN (
                SELECT id FROM (
                    SELECT 
                        id,
                        report_period,
                        ROW_NUMBER() OVER (
                            PARTITION BY company_id, LEFT(CAST(report_period AS VARCHAR), 4) 
                            -- 按报告期降序，确保最新的季度排在 rn=1
                            ORDER BY report_period DESC
                        ) as rn,
                        LEFT(CAST(report_period AS VARCHAR), 4) as report_year
                    FROM faf_related_party
                    WHERE delete_identify = 0 
                      AND company_id = #{companyId}
                      AND report_period &gt;= CAST(CONCAT(#{startYear}, '11') AS INT)
                      AND report_period &lt;= CAST(CONCAT(#{endYear}, '41') AS INT)
                      AND (
                          -- 最新的一年（这里取 endYear，假设用户查的就是截止 endYear 的数据）
                          LEFT(CAST(report_period AS VARCHAR), 4) = CAST(#{endYear} AS VARCHAR)
                          OR
                          -- 历年：只取年报 (Q4)
                          CAST(report_period AS VARCHAR) LIKE '%41'
                      )
                ) t
                WHERE 
                    -- 最新的一年里，只取排在最前面的（即季度最大的 rn=1）
                    (report_year = CAST(#{endYear} AS VARCHAR) AND rn = 1)
                    OR
                    -- 历年的年报，全部保留
                    report_year &lt; CAST(#{endYear} AS VARCHAR)
            )
        </when>
    </choose>

    <!-- 下面是原有的动态过滤条件，完美保留 -->
    <if test="relatedPartyNameList != null and relatedPartyNameList.size() > 0">
        AND related_party_name IN
        <foreach collection="relatedPartyNameList" item="name" open="(" separator="," close=")">
            #{name}
        </foreach>
    </if>

    <if test="relatedPartyRelationList != null and relatedPartyRelationList.size() > 0">
        AND related_party_relation IN
        <foreach collection="relatedPartyRelationList" item="name" open="(" separator="," close=")">
            #{name}
        </foreach>
    </if>
</select>
