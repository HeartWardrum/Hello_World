import pandas as pd
import re

# 模拟你的原始数据
data = {
    'sql_type': [
        'bigint', 
        'varchar(100)', 
        'decimal(38,6)', 
        'int identity(1,1)', 
        'nvarchar(max)',
        'numeric(10,2)',
        'char(10)'
    ]
}
df = pd.DataFrame(data)

# 函数 1：提取长度 (逗号左边的数字或唯一数字)
def extract_length(sql_type):
    s = str(sql_type).strip()
    # 匹配括号，并尝试抓取逗号前的部分 (.*?)(?:,.*)?
    match = re.search(r'\((.*?)(?:,.*)?\)', s)
    if match:
        res = match.group(1).strip()
        # 如果是 max，达梦不支持 max 关键字，通常可以留空或者转为数字，这里先按你要求保留或留空
        return res
    return None

# 函数 2：提取精度 (逗号右边的数字)
def extract_scale(sql_type):
    s = str(sql_type).strip()
    # 匹配括号中带有逗号的内容，并抓取逗号后面的数字
    match = re.search(r'\(.*?,\s*(.*?)\)', s)
    if match:
        return match.group(1).strip()
    return None

# 应用到 DataFrame 中
df['field_length'] = df['sql_type'].apply(extract_length)
df['field_scale'] = df['sql_type'].apply(extract_scale)

# 打印查看结果
print(df)
