WITH IndexDetails AS (
    SELECT 
        -- 💡 新增：提取并转换模式名
        LOWER(s.name COLLATE database_default) AS SchemaName,
        LOWER(t.name COLLATE database_default) AS TableName,
        LOWER(i.name COLLATE database_default) AS IndexName,
        CASE 
            WHEN i.type_desc = 'CLUSTERED' THEN 'CLUSTER'
            ELSE '' 
        END AS IndexType,
        i.is_unique AS IsUnique,
        STUFF((
            SELECT ', ' + LOWER(c.name COLLATE database_default) + CASE WHEN ic.is_descending_key = 1 THEN ' DESC' ELSE '' END
            FROM sys.index_columns ic
            JOIN sys.columns c ON ic.object_id = c.object_id AND ic.column_id = c.column_id
            WHERE ic.object_id = i.object_id AND ic.index_id = i.index_id AND ic.is_included_column = 0
            ORDER BY ic.key_ordinal
            FOR XML PATH('')
        ), 1, 2, '') AS KeyColumns,
        
        ROW_NUMBER() OVER (
            PARTITION BY t.object_id 
            ORDER BY i.name COLLATE database_default
        ) AS GroupSeqRaw
        
    FROM sys.indexes i
    JOIN sys.objects t ON i.object_id = t.object_id
    -- 💡 核心改动：关联 sys.schemas 表获取模式名
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    WHERE t.is_ms_shipped = 0 
      AND t.type COLLATE database_default = 'U' 
      AND i.type > 0 
      AND i.is_primary_key = 0 
)
SELECT 
    SchemaName AS [模式名],
    TableName AS [表名],
    RIGHT('0' + CAST(GroupSeqRaw AS VARCHAR(2)), 2) AS [组内序号],
    IndexName AS [索引名],
    KeyColumns AS [涉及字段名],
    -- 💡 核心改动：在 ON 后面加上 SchemaName.TableName
    'CREATE ' + 
    CASE WHEN IsUnique = 1 THEN 'UNIQUE ' ELSE '' END + 
    CASE WHEN IndexType <> '' THEN IndexType + ' ' ELSE '' END + 
    'INDEX ' + IndexName + ' ON ' + SchemaName + '.' + TableName + ' (' + KeyColumns + ');' AS [达梦建索引SQL语句]
FROM IndexDetails
ORDER BY SchemaName, TableName, [组内序号];
