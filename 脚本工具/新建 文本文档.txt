import numpy as np

# 假设你已经从数据库或文件中读取了这两个表：
# df_t  对应 financialschema.dwd_lc_relationship (公司关联关系明细)
# df_t1 对应 financialschema.dwd_CT_SystemConst (系统常量表)

# ==========================================
# 第一步：数据过滤 (提前过滤可以提高 Merge 效率)
# ==========================================

# 1. 过滤 t 表 (dwd_lc_relationship)
# 对应: where t.RelatedPartyCode is not null and t.CompanyCode <> t.RelatedPartyCode
cond_t = df_t['RelatedPartyCode'].notna() & (df_t['CompanyCode'] != df_t['RelatedPartyCode'])
t_filtered = df_t[cond_t].copy()

# 2. 过滤 t1 表 (dwd_CT_SystemConst)
# 对应: and t1.DM IN (12,13,14,15,16,17,22) and t1.LB = 2067
# 这里用到了你上一问刚学的 .isin()
cond_t1 = df_t1['DM'].isin([12, 13, 14, 15, 16, 17, 22]) & (df_t1['LB'] == 2067)
t1_filtered = df_t1[cond_t1].copy()


# ==========================================
# 第二步：表连接 (Inner Join)
# ==========================================
# 对应: inner join ... on t.Relationship = t1.DM
merged_df = pd.merge(
    t_filtered, 
    t1_filtered, 
    left_on='Relationship', 
    right_on='DM', 
    how='inner'
)


# ==========================================
# 第三步：实现窗口函数去重 (Row_Number & rn=1)
# ==========================================
# 对应: row_number() over(partition by CompanyCode, RelatedPartyCode order by isnull(...) desc) ... where rn=1

# 1. 处理 order by 里的 isnull(t.InfoPublDate, '1970-01-01...')
merged_df['InfoPublDate_sort'] = merged_df['InfoPublDate'].fillna('1970-01-01 00:00:00')

# 2. 排序 (按照 partition by 的字段和 order by 的字段排序)
# partition 字段默认升序，InfoPublDate 要求 desc(降序)，所以 ascending 分别设为 True, True, False
merged_df = merged_df.sort_values(
    by=['CompanyCode', 'RelatedPartyCode', 'InfoPublDate_sort'], 
    ascending=[True, True, False]
)

# 3. 取每个分组的第一条 (等价于 where rn = 1)
# 按照 partition 字段去重，保留排在最前面的第一项
final_df = merged_df.drop_duplicates(subset=['CompanyCode', 'RelatedPartyCode'], keep='first')


# ==========================================
# 第四步：提取最终字段并转换类型
# ==========================================
# 对应: cast(... as varchar) as company_id 等

# 构建最终的临时表 (等价于 into #tmp_pnsi_03_rel_ent)
tmp_pnsi_03_rel_ent = pd.DataFrame({
    # 使用 .astype(str) 模拟 cast(... as varchar)
    'company_id': final_df['CompanyCode'].astype(str),
    'rel_company_id': final_df['RelatedPartyCode'].astype(str),
    'rel_company_name': final_df['RelatedPartyName'],
    'relationship_description': final_df['MS']
})

# 重置索引（保持数据干净）
tmp_pnsi_03_rel_ent = tmp_pnsi_03_rel_ent.reset_index(drop=True)
