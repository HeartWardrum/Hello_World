# 降维打击：全部转成字符串、去空格、排序
def force_compare(df1, df2):
    def clean_df(df):
        # 转成字符串 -> 去除两端空格 -> 排序列 -> 重置索引
        d = df.astype(str).apply(lambda x: x.str.strip() if x.dtype == "object" else x)
        d = d.reindex(columns=sorted(d.columns))
        return d.sort_values(by=list(d.columns)).reset_index(drop=True)

    return clean_df(df1).equals(clean_df(df2))

# 试一下这个，看看是不是变 True 了？
print(force_compare(df1, df2))
