# 降维打击：全部转成字符串、去空格、排序
def force_compare(df1, df2):
    def clean_df(df):
        # 转成字符串 -> 去除两端空格 -> 排序列 -> 重置索引
        d = df.astype(str).apply(lambda x: x.str.strip() if x.dtype == "object" else x)
        d = d.reindex(columns=sorted(d.columns))
        return d.sort_values(by=list(d.columns)).reset_index(drop=True)

    return clean_df(df1).equals(clean_df(df2))

# 试一下这个，看看是不是变 True 了？
print(force_compare(df1, df2))


import pandas as pd

def force_compare_and_show_diff(df1, df2):
    # 1. 沿用你的标准化清洗逻辑
    def clean_df(df):
        # 注意：这里稍微改了一下，既然全部 astype(str) 了，所有列的 dtype 都会变成 object
        # 所以直接 x.str.strip() 即可
        d = df.astype(str).apply(lambda x: x.str.strip())
        return d.reindex(columns=sorted(d.columns))

    clean1 = clean_df(df1)
    clean2 = clean_df(df2)

    # 2. 判断两边列名是否一致，如果列名都不一样，先提醒
    if list(clean1.columns) != list(clean2.columns):
        print("⚠️ 警告：两个 DataFrame 的列名不一致！")
        print(f"df1 的列: {list(clean1.columns)}")
        print(f"df2 的列: {list(clean2.columns)}")
        return False

    # 3. 核心：使用 merge 找出差异
    # how='outer' 保留两边所有的数据
    # indicator=True 会多出一列叫 '_merge'，值分别为 'left_only', 'right_only', 'both'
    merged = pd.merge(clean1, clean2, how='outer', indicator=True)

    # 4. 提取差异
    left_only = merged[merged['_merge'] == 'left_only'].drop(columns=['_merge'])
    right_only = merged[merged['_merge'] == 'right_only'].drop(columns=['_merge'])

    # 5. 打印结果
    if left_only.empty and right_only.empty:
        print("✨ 完美匹配！两个 DataFrame 完全相同。")
        return True
    else:
        print("❌ 发现不一致！差异如下：\n")
        
        if not left_only.empty:
            print(f"💡 【只在 df1 中存在】的行（或 df2 中没有/少掉的行），共 {len(left_only)} 行：")
            print(left_only)
            print("-" * 50)
            
        if not right_only.empty:
            print(f"💡 【只在 df2 中存在】的行（或 df1 中多出的行），共 {len(right_only)} 行：")
            print(right_only)
            print("-" * 50)
            
        return False

# 调试时直接调用这个函数即可
force_compare_and_show_diff(df1, df2)
