def check_equal_totally_ignore_order(df1, df2):
    # 1. 如果行列数量都不等，直接返回 False
    if df1.shape != df2.shape:
        return False
    
    # 2. 将两者的列名按字母顺序横向排序（对齐列序）
    df1_aligned = df1.reindex(columns=sorted(df1.columns))
    df2_aligned = df2.reindex(columns=sorted(df2.columns))
    
    # 3. 把所有的列都当作排序键，进行纵向排序（对齐行序）
    # sort_values(by=list(...)) 会从第一列开始比，一样就比第二列，以此类推
    df1_sorted = df1_aligned.sort_values(by=list(df1_aligned.columns)).reset_index(drop=True)
    df2_sorted = df2_aligned.sort_values(by=list(df2_aligned.columns)).reset_index(drop=True)
    
    # 4. 此时两者的行列顺序已完全一致，直接对比内容
    return df1_sorted.equals(df2_sorted)
