WITH IndexDetails AS (
    SELECT 
        t.name COLLATE database_default AS TableName,
        i.name COLLATE database_default AS IndexName,
        -- 将 SQL Server 的索引类型翻译为达梦合法关键字
        CASE 
            WHEN i.type_desc = 'CLUSTERED' THEN 'CLUSTER'
            ELSE '' 
        END AS IndexType,
        i.is_unique AS IsUnique,
        -- 1. 拼接键列：去掉中括号，去掉 ASC，保留 DESC
        STUFF((
            SELECT ', ' + c.name COLLATE database_default + CASE WHEN ic.is_descending_key = 1 THEN ' DESC' ELSE '' END
            FROM sys.index_columns ic
            JOIN sys.columns c ON ic.object_id = c.object_id AND ic.column_id = c.column_id
            WHERE ic.object_id = i.object_id AND ic.index_id = i.index_id AND ic.is_included_column = 0
            ORDER BY ic.key_ordinal
            FOR XML PATH('')
        ), 1, 2, '') AS KeyColumns,
        
        -- 💡 核心新增：按表名分组，在组内对索引名排序，生成从 1 开始的序号
        ROW_NUMBER() OVER (
            PARTITION BY t.object_id 
            ORDER BY i.name COLLATE database_default
        ) AS GroupSeqRaw
        
    FROM sys.indexes i
    JOIN sys.objects t ON i.object_id = t.object_id
    WHERE t.is_ms_shipped = 0 
      AND t.type COLLATE database_default = 'U' 
      AND i.type > 0 
      AND i.is_primary_key = 0 
)
SELECT 
    TableName AS [表名],
    -- 💡 核心新增：将原生序号格式化为双位数（如 01, 02, 11）
    RIGHT('0' + CAST(GroupSeqRaw AS VARCHAR(2)), 2) AS [组内序号],
    IndexName AS [索引名],
    KeyColumns AS [涉及字段名],
    -- 动态组合成符合达梦（DM）语法的 CREATE INDEX 语句
    'CREATE ' + 
    CASE WHEN IsUnique = 1 THEN 'UNIQUE ' ELSE '' END + 
    CASE WHEN IndexType <> '' THEN IndexType + ' ' ELSE '' END + 
    'INDEX ' + IndexName + ' ON ' + TableName + ' (' + KeyColumns + ');' AS [达梦建索引SQL语句]
FROM IndexDetails
ORDER BY TableName, [组内序号];
