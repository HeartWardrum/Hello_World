import pandas as pd

data = {
    'column_name': ['id', 'name', 'remark'],
    'sql_type': ['int', 'varchar(100)', 'nvarchar(200)']
}
df = pd.DataFrame(data)

# 定义转换逻辑的函数
def convert_to_dm(sql_type):
    # 转换为小写字符串，防止大小写不一致
    sql_type_lower = str(sql_type).lower()
    
    if 'varchar' in sql_type_lower:
        return 'VARCHAR'
    elif 'int' in sql_type_lower:
        return 'INT'
    elif 'datetime' in sql_type_lower:
        return 'TIMESTAMP'
    # 如果没匹配到，保留原样或返回自定义默认值
    return sql_type.upper() 

# 运用 apply 函数应用到整列
df['dm_type'] = df['sql_type'].apply(convert_to_dm)

print(df)
