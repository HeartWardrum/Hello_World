#!/bin/bash
# 文件名：check_java_container.sh
# 用途：列出所有 Java 进程对应的容器名和启动方式

for pid in $(pgrep java 2>/dev/null); do
    echo "------------------------------------------------"

    # 从 cgroup 提取容器 ID（Podman/Docker 都适用）
    cid=$(cat /proc/"$pid"/cgroup 2>/dev/null \
          | grep -oP '(?<=/)[0-9a-f]{12,64}' \
          | tail -1)

    # 用容器 ID 反查容器名
    if [ -n "$cid" ]; then
        # 先试 Podman，再试 Docker
        name=$(podman inspect --format '{{.Name}}' "$cid" 2>/dev/null | sed 's|^/||')
        [ -z "$name" ] && name=$(docker inspect --format '{{.Name}}' "$cid" 2>/dev/null | sed 's|^/||')
        [ -z "$name" ] && name="(容器 $cid)"
    else
        name="(非容器进程)"
    fi

    # 启动方式（从容器内 Java 进程的命令行取）
    cmd=$(tr '\0' ' ' < /proc/"$pid"/cmdline 2>/dev/null)
    if echo "$cmd" | grep -q -- '-jar '; then
        jar=$(echo "$cmd" | grep -oP '(?<=-jar )\S+\.jar' | head -1)
        how="java -jar $jar"
    elif echo "$cmd" | grep -qE -- '-cp | -classpath '; then
        cp=$(echo "$cmd" | grep -oP '(?<=-cp | -classpath )\S+' | head -1)
        how="java -cp $cp"
    else
        how="其他: $cmd"
    fi

    echo "PID      : $pid"
    echo "容器名   : $name"
    echo "启动方式 : $how"
    echo ""
done
