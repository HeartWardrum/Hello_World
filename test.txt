#!/bin/bash
# 文件名：check_java_service.sh
# 用途：只列出 Java 进程的服务名和启动方式

for pid in $(pgrep java 2>/dev/null); do
    # 服务名（systemd）
    svc=$(systemctl status "$pid" 2>/dev/null | head -1 | awk '{print $2}')
    [ -z "$svc" ] && svc="(非 systemd 管理)"

    # 启动命令
    cmd=$(tr '\0' ' ' < /proc/"$pid"/cmdline 2>/dev/null)

    # 启动方式
    if echo "$cmd" | grep -q -- '-jar '; then
        jar=$(echo "$cmd" | grep -oP '(?<=-jar )\S+\.jar' | head -1)
        how="java -jar $jar"
    elif echo "$cmd" | grep -qE -- '-cp | -classpath '; then
        cp=$(echo "$cmd" | grep -oP '(?<=-cp | -classpath )\S+' | head -1)
        how="java -cp $cp"
    else
        how="其他: $cmd"
    fi

    echo "PID: $pid"
    echo "  服务名  : $svc"
    echo "  启动方式: $how"
    echo ""
done
