#!/bin/bash
# 文件名：check_java_realname.sh
# 用途：从父进程链和工作目录推断 Java 程序的真实身份

for pid in $(pgrep java 2>/dev/null); do
    echo "------------------------------------------------"

    # 工作目录
    cwd=$(readlink /proc/"$pid"/cwd 2>/dev/null)

    # 父进程链：逐级往上找，直到 PID 1
    chain=""
    cur=$pid
    for i in 1 2 3 4 5; do
        ppid=$(ps -o ppid= -p "$cur" 2>/dev/null | tr -d ' ')
        [ -z "$ppid" ] || [ "$ppid" == "0" ] || [ "$ppid" == "1" ] && break
        pcmd=$(tr '\0' ' ' < /proc/"$ppid"/cmdline 2>/dev/null)
        chain="$chain -> [$ppid] $pcmd"
        cur=$ppid
    done

    # 启动命令
    cmd=$(tr '\0' ' ' < /proc/"$pid"/cmdline 2>/dev/null)

    # 启动方式
    if echo "$cmd" | grep -q -- '-jar '; then
        jar=$(echo "$cmd" | grep -oP '(?<=-jar )\S+\.jar' | head -1)
        how="java -jar $jar"
    elif echo "$cmd" | grep -qE -- '-cp | -classpath '; then
        cp=$(echo "$cmd" | grep -oP '(?<=-cp | -classpath )\S+' | head -1)
        how="java -cp $cp"
    else
        how="其他"
    fi

    echo "PID      : $pid"
    echo "工作目录 : $cwd"
    echo "启动方式 : $how"
    echo "父进程链 :$chain"
    echo ""
done
